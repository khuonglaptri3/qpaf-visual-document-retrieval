## Review Focus

1. **Modal auto-retries an OOM-killed input.** Modal docs: "Modal automatically retries any inputs that a container was running when it failed." The above-limit control may run in more than one container. If any attempt *completes*, enforcement is not demonstrated. Pinned by `test_above_limit_completion_on_retry_is_not_demonstrated` (Task 1).
2. **The above-limit control fails for a non-memory reason** (timeout, watchdog, image error). That is `INCONCLUSIVE`, never `ENFORCEMENT_DEMONSTRATED`, so M3.6 stays blocked. Pinned by `test_failure_without_oom_attribution_is_inconclusive` (Task 1).
3. **Evidence at exact threshold boundaries** (mean `0.01`, CI lower `0.0`, share `0.90`, latency `10.0`, CUDA `1.5`). Inclusive and exclusive bounds must match the protocol. Pinned by `test_threshold_boundaries` (Task 3).
4. **An evidence file is edited after pinning, a field is NaN or missing, or an M3.6 acceptance record is absent.** The result must be a refusal or `BLOCKED`, never a quality-based decision. Pinned by `test_pin_drift_refuses`, `test_missing_or_nonfinite_evidence_blocks` and `test_unaccepted_prerequisite_blocks_strong_quality` (Task 3).
5. **R3's evidence index or links go stale** after a later commit touches a cited file. Pinned by `test_evidence_index_hashes_match` and `test_relative_links_resolve` (Task 5).

```python
class MemoryProbeTests(unittest.TestCase):
    def setUp(self):
        self.api = importlib.import_module('scripts.m36_memory_probe')
        self.ok = {'status': 'COMPLETED', 'target_mib': 2048, 'error_type': None, 'error_text': None, 'in_sandbox': {}}
        self.oom = {'status': 'FAILED', 'target_mib': 4608, 'error_type': 'FunctionError',
                    'error_text': 'Container ran out of memory (OOM)', 'in_sandbox': None}
        self.stats = {'input_success_count': 1, 'input_failure_count': 1,
                      'container_total_count': 2, 'container_error_count': 1}

    def test_constants_match_reviewed_design(self):
        self.assertEqual((self.api.PROBE_LIMIT_MIB, self.api.BELOW_TARGET_MIB, self.api.ABOVE_TARGET_MIB),
                         (4096, 2048, 4608))

    def test_touch_allocation_commits_requested_bytes(self):
        result = self.api.touch_allocation(8)
        self.assertEqual(result['allocated_bytes'], 8 * 2**20)
        if os.name != 'nt':  # resource.getrusage is Linux-only; the probe always runs on Linux
            self.assertGreater(result['ru_maxrss_bytes'], 0)

    def test_demonstrated_requires_below_success_and_attributed_above_oom(self):
        self.assertEqual(self.api.classify_controls(self.ok, self.oom, self.stats), 'ENFORCEMENT_DEMONSTRATED')

    def test_above_limit_completion_on_retry_is_not_demonstrated(self):
        stats = {**self.stats, 'input_success_count': 2, 'container_total_count': 3}
        self.assertEqual(self.api.classify_controls(self.ok, self.oom, stats), 'ENFORCEMENT_NOT_DEMONSTRATED')
        self.assertEqual(self.api.classify_controls(self.ok, {**self.ok, 'target_mib': 4608}, self.stats),
                         'ENFORCEMENT_NOT_DEMONSTRATED')

    def test_failure_without_oom_attribution_is_inconclusive(self):
        timeout = {**self.oom, 'error_type': 'TimeoutError', 'error_text': 'watchdog stopped app'}
        self.assertEqual(self.api.classify_controls(self.ok, timeout, {**self.stats, 'container_error_count': 0}),
                         'INCONCLUSIVE')
        self.assertEqual(self.api.classify_controls({**self.oom, 'target_mib': 2048}, self.oom, self.stats),
                         'INCONCLUSIVE')

    def test_oom_attribution_reads_error_text_or_provider_reason(self):
        self.assertTrue(self.api.oom_attributed('Killed: exit code 137', None))
        self.assertTrue(self.api.oom_attributed('', {'exit_reasons': ['OOM']}))
        self.assertFalse(self.api.oom_attributed('ImportError', {'exit_reasons': []}))

    def test_bound_verdict_only_when_demonstrated(self):
        runs = {'A02': {'limit_mib': 4096, 'success_marker': True, 'failed_calls': 0},
                'A03': {'limit_mib': 5120, 'success_marker': True, 'failed_calls': 0}}
        good = self.api.memory_bound_verdict('ENFORCEMENT_DEMONSTRATED', runs)
        self.assertEqual(good['issue_status'], 'RESOLVED_BY_ENFORCED_BOUND')
        self.assertEqual(good['runs']['A02'], {'bound': 'BELOW_LIMIT', 'limit_mib': 4096})
        self.assertFalse(good['exact_peak_available'])
        bad = self.api.memory_bound_verdict('INCONCLUSIVE', runs)
        self.assertEqual(bad['issue_status'], 'OPEN')
        failed = self.api.memory_bound_verdict('ENFORCEMENT_DEMONSTRATED',
                                               {**runs, 'A02': {**runs['A02'], 'failed_calls': 1}})
        self.assertEqual(failed['runs']['A02']['bound'], 'UNBOUNDED')
```

`g2_outcome` precedence, which sets the order the tests rely on:
1. **BLOCKED:** a prerequisite is not `ACCEPTED`/`PASS`; `query_count != wtl_total`; `test_labels_opened`; replay difference `> 0` or `not m35_replay_equal`; any required field missing or non-finite.
2. **NO-GO-PIVOT:** `mean_delta < 0.01` or `ci95_lower <= 0`.
3. **PARTIAL-INVESTIGATE:** `top5_share >= 0.90`, or any operational failure (`median_latency_ms > 10.0`, `peak_cuda_gib >= 1.5`, `reload_max_abs_diff > 1e-7`, `not finite_outputs`).
4. **GO.**

Threshold values come from `criteria`, never from literals in the function.

```python
CRITERIA = {'mean_delta_min_inclusive': 0.01, 'ci95_lower_min_exclusive': 0.0,
            'top5_share_max_exclusive': 0.90, 'latency_ms_max_inclusive': 10.0,
            'cuda_gib_max_exclusive': 1.5, 'reload_abs_tolerance': 1e-7}

def evidence(**over):
    base = {'prerequisites': {'M3.3': 'ACCEPTED', 'M3.5': 'ACCEPTED', 'M3.6_technical': 'PASS',
                              'M3.6_owner': 'ACCEPTED', 'M3.6_qa': 'ACCEPTED'},
            'pairing': {'query_count': 171, 'wtl_total': 171, 'test_labels_opened': False,
                        'metric_replay_max_abs_difference': 0.0, 'm35_replay_equal': True},
            'quality': {'mean_delta': 0.114132, 'ci95_lower': 0.077501, 'top5_share': 0.2809},
            'operational': {'median_latency_ms': 2.447, 'peak_cuda_gib': 0.0325,
                            'reload_max_abs_diff': 0.0, 'finite_outputs': True}}
    for path, value in over.items():
        section, key = path.split('__')
        base[section] = {**base[section], key: value}
    return base

class G2OutcomeTests(unittest.TestCase):
    def test_go_with_current_evidence_shape(self):
        result = g2_outcome(evidence(), CRITERIA)
        self.assertEqual((result['decision'], result['m4_branch']), ('GO', 'continue_qpaf'))

    def test_threshold_boundaries(self):
        self.assertEqual(g2_outcome(evidence(quality__mean_delta=0.01), CRITERIA)['decision'], 'GO')
        self.assertEqual(g2_outcome(evidence(quality__ci95_lower=0.0), CRITERIA)['decision'], 'NO-GO-PIVOT')
        self.assertEqual(g2_outcome(evidence(quality__top5_share=0.90), CRITERIA)['decision'], 'PARTIAL-INVESTIGATE')
        self.assertEqual(g2_outcome(evidence(operational__median_latency_ms=10.0), CRITERIA)['decision'], 'GO')
        self.assertEqual(g2_outcome(evidence(operational__peak_cuda_gib=1.5), CRITERIA)['decision'], 'PARTIAL-INVESTIGATE')
        self.assertEqual(g2_outcome(evidence(operational__reload_max_abs_diff=2e-7), CRITERIA)['decision'], 'PARTIAL-INVESTIGATE')

    def test_negative_quality_is_no_go_even_if_operational_fails(self):
```
